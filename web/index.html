MiniLang 2.0 Syntax Quick Reference

Dictionaries (HashMaps)

let user = { "name": "Alice", "age": 20 };
user["role"] = "Admin";
print(has(user, "name"));
print(keys(user));

Try / Catch Exception Handling

try {
  let x = 10 / 0;
} catch (err) {
  print("Caught error: " + err);
}

Multi-line Strings

let banner = """
╔═════════════════╗
║ MiniLang 2.0 ║
╚═════════════════╝
""";
print(banner);

Kid-Friendly English Syntax

let count = 0;
add 5 to count;
multiply count by 2;
repeat 3 times {
  say "Hello!";
}

Object-Oriented Classes

class Point {
  init(x, y) {
    this.x = x;
    this.y = y;
  }
  show() {
    print("(" + str(this.x) + ")");
  }
}
