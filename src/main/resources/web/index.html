MiniLang 2.0 Syntax Reference

Kid-Friendly English Syntax

score = 10;
score = score + 5;
score = score * 2;
show "Final: " + str(score);
repeat 3 times {
  say "Hello";
}

Object-Oriented Classes

class Point {
  init(x, y) {
    this.x = x;
    this.y = y;
  }
  show() {
    print("(" + str(this.x) + ")");
  }
}

Try / Catch Exception Handling

try {
  let x = 10 / 0;
} catch (err) {
  print("Caught error: " + err);
}
